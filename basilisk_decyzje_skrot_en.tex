\begin{figure}[H]

\begin{tikzpicture}[node distance=9mm and 6mm]

  \node[rd=5.2cm, below=14mm of wz] (rd) {\textbf{Board of Directors}\\3--7 directors, 3-year term\\Chair: casting vote};
  \node[dw=3.8cm, below=14mm of rd, xshift=-2.6cm] (dw) {\textbf{Executive director}\\(at least one)\\day-to-day business};
  \node[rd=3.8cm, below=14mm of rd, xshift=2.6cm] (dn) {\textbf{Non-executive director}\\(at least one)\\supervision, agreements with exec.~dir.};

  \draw[line] (akc) -- (wz);
  \draw[line] (wz) -- node[lbl, right, text width=4.2cm, align=left]{appoints and dismisses\\(absolute majority of votes cast)} (rd);
  \draw[line] (rd.south) ++(-1.2,0) -- node[lbl, left]{delegates} (dw.north);
  \draw[line] (rd.south) ++(1.2,0) -- (dn.north);
  \draw[line] (rd) -- node[lbl, above, text width=1.9cm]{consent of all directors} (pr);
  \node[box=10cm, below=8mm of dn.south west, anchor=north, xshift=-0.8cm, fill=black!3] (rep) {\textbf{Representation towards third parties:} two directors jointly or a director with a commercial proxy (Art.~\S~\ref{en-art:reprezentacja}(1))};
\end{tikzpicture}
\caption{Structure of the corporate bodies (Art.~\S\S~\ref{en-art:rada}--\ref{en-art:wz}).}
\end{figure}
\begin{Komentarz}
Shareholders elect the Board; the Board manages and supervises, and the executive director runs day-to-day business. Two directors, or a director with a commercial proxy, sign for the Company (Art.~\S~\ref{en-art:reprezentacja}).
\end{Komentarz}

\begin{figure}[H]

\begin{tikzpicture}[node distance=7mm and 9mm]

  \node[q, below=of q1] (q2) {Matter on the\\``75\%'' list (table 1)?};
  \node[wz=4.4cm, right=of q2] (a2) {\textbf{GM --- 75\% of all votes in the Company}\\Art.~\S~\ref{en-art:wz-zastrzezone}(1)};
  \node[q, below=of q2] (q3) {Directors, financial statements, discharge, profit, directors' pay?};

  \node[q, below=of q3] (q4) {Matter on the Board list (table 2)?};
  \node[rd=4.4cm, right=of q4] (a4) {\textbf{Resolution of the Board of Directors}\\Art.~\S~\ref{en-art:rada}(5) and (7)};
  \node[dw=4.4cm, below=of q4] (a5) {\textbf{Executive director}\\within the budget and the Board Rules};

\begin{Komentarz}
Start at the top: the first “yes” gives the body. “75\% of all votes” counts every share, not only those present --- an absent shareholder counts as “against”.
\end{Komentarz}

\end{Komentarz}

\end{Komentarz}

\begin{Komentarz}
With a 50/20/10/10/10 split: Founder~A (50\%) cannot amend the Articles alone but can block any amendment; A and B (70\%) are not enough --- at least one more Founder with 10\% is needed. No amendment is possible without A (the others hold 50\% together).
\end{Komentarz}
