\chapter*{Projekt umowy prostej spółki akcyjnej --- wariant prosty}
\addcontentsline{toc}{chapter}{Projekt umowy prostej spółki akcyjnej --- wariant prosty}

\begin{center}
{\Large\bfseries UMOWA PROSTEJ SPÓŁKI AKCYJNEJ}\par
\vspace{0.25cm}
{\LARGE\bfseries \CompanyFull}\par
\vspace{0.6cm}
\end{center}

Stawający, wskazani w Załączniku nr~1 do niniejszej umowy, zwani dalej „Założycielami”, zawiązują prostą spółkę akcyjną na następujących zasadach.

Niniejsza umowa reguluje wyłącznie to, co musi albo powinno znaleźć się w umowie spółki: ustrój Spółki, akcje, limity emisji, tryb podejmowania uchwał i ograniczenia zbywania akcji. Pozostałe kwestie --- w szczególności zasady współpracy Założycieli, polityki bezpieczeństwa i zgodności, regulamin Rady Dyrektorów, regulamin Walnego Zgromadzenia i regulamin programu motywacyjnego --- regulują odrębne dokumenty przyjmowane uchwałami organów, bez zmiany niniejszej umowy.

\begin{ContractClause}{Firma, siedziba i czas trwania}
\label{art:firma}
\begin{enumerate}
  \item Firma Spółki brzmi: „\CompanyFull”. Spółka może posługiwać się skrótem „\CompanyShort” oraz oznaczeniami handlowymi i znakami towarowymi, w szczególności oznaczeniem „Basilisk Systems”.
  \item Siedzibą Spółki jest Pruszków.
  \item Spółka może prowadzić działalność w Rzeczypospolitej Polskiej i za granicą oraz tworzyć oddziały, zakłady, laboratoria, przedstawicielstwa i spółki zależne, a także uczestniczyć w innych podmiotach.
  \item Spółka zostaje zawiązana na czas nieoznaczony.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Przedmiot działalności i PKD 2025}
\label{art:pkd}
\begin{enumerate}
\item Przedmiotem działalności Spółki jest projektowanie, badanie, rozwój, produkcja, integracja, wdrażanie, sprzedaż i serwis fizycznej (ucieleśnionej) sztucznej inteligencji i systemów autonomicznych --- w szczególności oprogramowania autonomii, percepcji, koordynacji zespołowej, dowodzenia, łączności i nawigacji --- oraz wykorzystujących te technologie bezzałogowych statków powietrznych, w tym autonomicznych rojów dronów, robotów lądowych i jednostek nawodnych, wraz z infrastrukturą operatorską, obliczeniową i serwisową.
\item Przedmiot działalności Spółki według PKD~2025 obejmuje:

\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Akcje i kapitał akcyjny}
\label{art:akcje}
\begin{enumerate}
  \item Akcje Spółki nie mają wartości nominalnej, nie stanowią części kapitału akcyjnego i są niepodzielne.
  \item Na dzień zawiązania Spółki emituje się \textbf{1 000 000 (jeden milion)} zwykłych akcji serii~A, oznaczonych numerami od~A0000001 do~A1000000. Cena emisyjna jednej akcji serii~A wynosi \textbf{0,01 zł (jeden grosz)}.
  \item Każda akcja daje prawo do jednego głosu na Walnym Zgromadzeniu oraz równe prawo do dywidendy i do udziału w majątku pozostałym po likwidacji. Spółka nie emituje akcji uprzywilejowanych ani akcji o szczególnych uprawnieniach; ich ustanowienie wymaga zmiany niniejszej umowy.
  \item Wysokości kapitału akcyjnego nie określa się w niniejszej umowie; podlega ona ujawnieniu i zmianom zgodnie z przepisami prawa.
  \item Akcje są rejestrowane w rejestrze akcjonariuszy prowadzonym przez podmiot wybrany uchwałą akcjonariuszy; pierwszego wyboru dokonują Założyciele przy zawiązaniu Spółki. Spółka zgłasza do rejestru akcjonariuszy ograniczenia rozporządzania akcjami i obowiązki związane z akcjami wynikające z niniejszej umowy, w tym oznaczenie akcji objętych obowiązkiem zbycia (§~\ref{art:obowiazek-zbycia}), i aktualizuje te dane w terminach przewidzianych prawem.
  \item Każdy akcjonariusz podaje do rejestru akcjonariuszy adres poczty elektronicznej wraz ze zgodą na komunikację ze Spółką przy jego wykorzystaniu i aktualizuje ten adres oraz pozostałe dane podlegające wpisowi w terminie 7 dni od zmiany. Skutki braku aktualnych danych w rejestrze obciążają akcjonariusza w zakresie dopuszczalnym przez prawo.
  \item Spółka może nabywać akcje własne wyłącznie w przypadkach i granicach określonych w art.~300$^{47}$ KSH, na podstawie uchwały Walnego Zgromadzenia podjętej zgodnie z §~\ref{art:wz-zastrzezone} ust.~1. Uchwała określa co najmniej maksymalną liczbę nabywanych akcji, cel nabycia, maksymalną cenę oraz kwotę kapitału rezerwowego utworzonego na ten cel; łączna liczba akcji własnych nie może przekroczyć 25\% wszystkich akcji Spółki.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Emisje akcji i maksymalne liczby akcji}
\label{art:emisje}
\begin{enumerate}
  \item \textbf{Zasada maksymalnej liczby akcji.} Każda uchwała o emisji akcji, o emisji warrantów, opcji, instrumentów zamiennych albo innych praw do akcji określa maksymalną liczbę akcji, które mogą zostać wyemitowane na jej podstawie, oraz termin, po którym prawa z niej wygasają. Uchwała, która nie określa maksymalnej liczby akcji, nie stanowi podstawy emisji. Dotyczy to także każdego upoważnienia udzielonego Radzie Dyrektorów oraz każdego programu, planu i umowy przewidujących prawo do objęcia akcji.
  \item \textbf{Łączny limit.} Łączna liczba akcji wszystkich serii istniejących i objętych nierozliczonymi prawami do objęcia akcji nie może przekroczyć \textbf{2 000 000 (dwóch milionów)}. Przekroczenie tego limitu wymaga zmiany niniejszej umowy.
  \item \textbf{Limity poszczególnych serii.} Na podstawie art.~300$^{103}$ KSH niniejsza umowa przewiduje możliwość emisji, w jednej lub kilku emisjach, na podstawie uchwał podjętych nie później niż do dnia \textbf{31 grudnia 2036 r.}:
  \begin{enumerate}[label=\alph*)]
    \item nie więcej niż \textbf{250 000 (dwustu pięćdziesięciu tysięcy)} zwykłych akcji serii~F (§~\ref{art:seria-f}), przy czym łączna liczba istniejących akcji serii~F nie może przekroczyć 20\% wszystkich akcji Spółki istniejących bezpośrednio po danej emisji;
    \item nie więcej niż \textbf{176 471 (stu siedemdziesięciu sześciu tysięcy czterystu siedemdziesięciu jednej)} zwykłych akcji serii~P na potrzeby programu motywacyjnego (§~\ref{art:seria-p}).
  \end{enumerate}
  \item \textbf{Nowe serie.} Emisja akcji serii innej niż~A, F i~P, w szczególności serii inwestorskiej, wymaga zmiany niniejszej umowy określającej oznaczenie serii, maksymalną liczbę akcji tej serii i termin na podjęcie uchwał emisyjnych. Zmiana umowy i uchwała emisyjna mogą zostać podjęte na tym samym Walnym Zgromadzeniu, z tym że uchwała emisyjna jest skuteczna nie wcześniej niż zmiana umowy.
  \item \textbf{Kompetencja.} Uchwałę o emisji podejmuje wyłącznie Walne Zgromadzenie, większością określoną w §~\ref{art:wz-zastrzezone} ust.~1. Rada Dyrektorów wykonuje taką uchwałę w jej granicach --- wybiera osoby, do których kierowana jest oferta, składa oferty, zawiera umowy objęcia akcji, przyjmuje wkłady, wydaje dyspozycje wpisu do rejestru akcjonariuszy i dokonuje zgłoszeń rejestrowych --- bez odrębnej uchwały Walnego Zgromadzenia dla każdej transzy i bez prawa przekroczenia maksymalnej liczby akcji ani obniżenia ceny emisyjnej poniżej minimum określonego w uchwale.
  \item \textbf{Prawo poboru.} Pozbawienie dotychczasowych akcjonariuszy prawa poboru w całości albo w części wymaga uchwały podjętej większością co najmniej czterech piątych głosów oraz spełnienia pozostałych wymogów art.~300$^{106}$ KSH; uchwała może objąć wszystkie akcje emitowane na podstawie danej uchwały emisyjnej.
  \item Cena emisyjna akcji nie może być niższa niż 0,01 zł za akcję, chyba że bezwzględnie obowiązujące przepisy wymagają ceny wyższej.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Seria F --- Założyciele Funkcjonalni}
\label{art:seria-f}
\begin{enumerate}
  \item „Założycielem Funkcjonalnym Serii~F” może zostać osoba fizyczna, która po zawiązaniu Spółki obejmuje trwałą, strategiczną i osobistą odpowiedzialność za istotny obszar rozwoju Spółki, porównywalną co do znaczenia z rolą Założyciela Pierwotnego, i zobowiązuje się do długoterminowego, istotnego zaangażowania operacyjnego. Sam status pracownika, współpracownika, dyrektora albo doradcy nie jest wystarczający. Posiadanie akcji serii~A nie wyłącza możliwości uzyskania tego statusu.
  \item Nadanie statusu Założyciela Funkcjonalnego Serii~F wymaga uchwały Walnego Zgromadzenia podjętej większością 75\% wszystkich głosów. Uchwała określa rolę strategiczną, minimalny zakres zaangażowania, maksymalną liczbę akcji serii~F przewidzianych dla danej osoby oraz „Dzień Przyznania Serii~F”, od którego liczy się okres, o którym mowa w §~\ref{art:obowiazek-zbycia} ust.~2.
  \item Uchwała o nadaniu statusu ma charakter kwalifikacyjny: nie stanowi oferty objęcia akcji ani prawa do objęcia akcji. Roszczenie o objęcie akcji serii~F powstaje wyłącznie z umowy objęcia akcji. Przed złożeniem przez Spółkę wiążącej oferty objęcia akcji Walne Zgromadzenie może, większością 75\% wszystkich głosów, cofnąć nadany status, jeżeli dana osoba przestała spełniać kryteria z ust.~1 albo warunki uchwały kwalifikacyjnej. Po powstaniu akcji ich los podlega wyłącznie przepisom prawa i niniejszej umowie; uchwała o utracie statusu nie pozbawia własności akcji.
  \item Każda emisja akcji serii~F wymaga odrębnej uchwały Walnego Zgromadzenia podjętej większością 75\% wszystkich głosów, określającej elementy wymagane przez art.~300$^{104}$ KSH, w granicach limitów z §~\ref{art:emisje} ust.~3 lit.~a. Jeżeli wkładem jest świadczenie pracy lub usług, uchwała i umowa objęcia określają rodzaj i czas świadczenia zgodnie z art.~300$^{104}$ §~1 pkt~7 KSH.
  \item Akcje serii~F są akcjami zwykłymi, równymi w prawach z akcjami serii~A.
  \item Od chwili powstania akcji serii~F stosuje się do nich §~\ref{art:zbywanie} i §~\ref{art:obowiazek-zbycia}; okres, o którym mowa w §~\ref{art:obowiazek-zbycia} ust.~2, liczy się od Dnia Przyznania Serii~F, a jeżeli akcje powstaną później --- z uwzględnieniem nieprzerwanego zaangażowania od tego dnia, z tym że żadne prawa z akcji nie powstają przed ich powstaniem.
  \item Seria~F jest odrębna od serii~P. Emisja akcji serii~F nie zmniejsza puli serii~P, a przyznanie akcji serii~P tej samej osobie nie obciąża limitu serii~F.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Seria P --- program motywacyjny}
\label{art:seria-p}
\begin{enumerate}
  \item Tworzy się pulę programu motywacyjnego dla pracowników, współpracowników, dyrektorów i doradców Spółki, odpowiadającą maksymalnie \textbf{176 471} akcji serii~P. Przy wykorzystaniu całej puli i braku innych emisji pula ta odpowiada 15\% akcji Spółki po pełnym rozwodnieniu; odstępstwo od dokładnych 15\% wynika z niepodzielności akcji.
  \item Akcje serii~P są emitowane na podstawie uchwał Walnego Zgromadzenia podjętych w granicach i terminie z §~\ref{art:emisje} ust.~3 lit.~b, w trybie zwykłym (art.~300$^{103}$--300$^{107}$ KSH) albo w trybie warunkowej emisji akcji (art.~300$^{114}$--300$^{118}$ KSH). Uchwała określa maksymalną liczbę akcji, cenę emisyjną albo sposób jej ustalenia, cel emisji, krąg osób uprawnionych i termin wykonania prawa do objęcia akcji; zgoda na zawarcie umów przyznających prawo do objęcia akcji może zostać udzielona w tej samej uchwale dla kategorii osób i na warunkach określonych w regulaminie programu.
  \item Zwiększenie puli ponad 176 471 akcji albo przedłużenie terminu, o którym mowa w §~\ref{art:emisje} ust.~3, wymaga zmiany niniejszej umowy.
  \item Rada Dyrektorów prowadzi program, rekomenduje uczestników i wykonuje uchwały emisyjne zgodnie z §~\ref{art:emisje} ust.~5; nie jest uprawniona do samodzielnej emisji akcji serii~P ani do pozbawienia prawa poboru.
  \item Przyznanie praw w programie nie daje statusu akcjonariusza do chwili powstania akcji i wpisu do rejestru akcjonariuszy. Warunki nabycia uprawnień, ograniczenia zbywania i skutki ustania współpracy określa regulamin programu i umowa uczestnictwa.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Zbywanie i obciążanie akcji}
\label{art:zbywanie}
\begin{enumerate}
  \item Zbycie albo obciążenie akcji wymaga zgody Spółki w rozumieniu art.~300$^{39}$ §~1 KSH. Zgody udziela albo odmawia jej Rada Dyrektorów w formie dokumentowej w terminie 14 dni od zgłoszenia Spółce zamiaru rozporządzenia; zgłoszenie wskazuje nabywcę, cenę i istotne warunki. \textbf{Jeżeli Rada Dyrektorów nie odmówi zgody w tym terminie, uważa się, że zgoda została udzielona.} Na żądanie akcjonariusza Spółka wydaje w terminie 7 dni zaświadczenie w formie dokumentowej stwierdzające udzielenie zgody albo jej powstanie z upływem terminu; zaświadczenie stanowi dokument uzasadniający wpis nabywcy w rejestrze akcjonariuszy.
  \item Rada Dyrektorów może odmówić zgody wyłącznie, jeżeli:
  \begin{enumerate}[label=\alph*)]
    \item nabywca albo podmiot go kontrolujący jest objęty sankcjami lub środkami ograniczającymi albo nabycie byłoby sprzeczne z przepisami o kontroli inwestycji, kontroli obrotu towarami o znaczeniu strategicznym albo o ochronie bezpieczeństwa państwa;
    \item nabycie wymaga zgody, braku sprzeciwu albo upływu terminu na sprzeciw właściwego organu, a warunek ten nie został spełniony --- w takim przypadku Rada Dyrektorów może zamiast odmowy udzielić zgody pod warunkiem zawieszającym jego spełnienia;
    \item rozporządzenie następowałoby w okresie, o którym mowa w ust.~4, bez wymaganej zgody; albo
    \item rozporządzenie dotyczy akcji objętych obowiązkiem zbycia (§~\ref{art:obowiazek-zbycia}), a nabywca nie przyjął na siebie obowiązków związanych z tymi akcjami, w tym nie udzielił pełnomocnictwa i nie złożył oferty, o których mowa w §~\ref{art:obowiazek-zbycia} ust.~6.
  \end{enumerate}
  Odmowa wskazuje konkretną podstawę z niniejszego ustępu. Z innych przyczyn odmowa zgody jest niedopuszczalna.
  \item Na podstawie art.~300$^{39}$ §~2 KSH strony postanawiają, że w razie odmowy zgody Spółka nie ma obowiązku wskazania innego nabywcy, a przepisów art.~300$^{39}$ §~3--5 KSH nie stosuje się. Akcjonariusz może wskazać innego nabywcę i ponowić zgłoszenie. Przepisu niniejszego nie stosuje się do zbycia akcji w postępowaniu egzekucyjnym (art.~300$^{39}$ §~6 KSH).
  \item Założyciel Pierwotny przez 12 miesięcy od dnia wpisu Spółki do rejestru przedsiębiorców, a Założyciel Funkcjonalny Serii~F przez 12 miesięcy od jego Dnia Przyznania Serii~F, nie może zbyć ani obciążyć akcji bez zgody Walnego Zgromadzenia podjętej większością 75\% wszystkich głosów. Ograniczenie to nie dotyczy zbycia w wykonaniu obowiązku zbycia (§~\ref{art:obowiazek-zbycia}) ani przeniesienia akcji do podmiotu w całości kontrolowanego przez tego akcjonariusza, jeżeli podmiot ten przyjmie na siebie obowiązki związane z akcjami, a przenoszący pozostanie solidarnie odpowiedzialny i zapewni zwrotne przeniesienie akcji przed utratą kontroli nad tym podmiotem.
  \item Zgody Spółki nie wymaga nabycie akcji w wykonaniu obowiązku zbycia (§~\ref{art:obowiazek-zbycia}) ani zbycie akcji w postępowaniu egzekucyjnym.
  \item Spółka nie przewiduje prawa pierwszeństwa, prawa przyłączenia do zbycia ani prawa przymusowego współzbycia akcji. Ustanowienie takich praw wymaga zmiany niniejszej umowy albo odrębnej umowy akcjonariuszy, która nie wiąże Spółki ani nabywców akcji niebędących jej stroną.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Obowiązek zbycia akcji po ustaniu zaangażowania}
\label{art:obowiazek-zbycia}
\begin{enumerate}
  \item Obowiązkiem zbycia objęte są akcje Założyciela Pierwotnego wskazanego w Załączniku nr~1 jako zobowiązanego do stałego, osobistego zaangażowania operacyjnego w rozwój Spółki oraz akcje serii~F Założyciela Funkcjonalnego Serii~F („Akcjonariusz Funkcjonalny”).
  \item Akcje zwalniają się z obowiązku zbycia przez 48 miesięcy od dnia wskazanego w Załączniku nr~1 albo od Dnia Przyznania Serii~F: przez pierwsze 12 miesięcy nie zwalnia się żadna akcja; po nieprzerwanym upływie 12 miesięcy zwalnia się jednorazowo 25\% akcji objętych obowiązkiem; przez kolejne 36 miesięcy akcje zwalniają się co miesiąc, w równych częściach, do zwolnienia 100\%.
  \item Do chwili zwolnienia akcje pozostają akcjami istniejącymi i dają pełne prawa korporacyjne i majątkowe; podlegają jednak obowiązkowi zbycia na zasadach niniejszego paragrafu.
  \item Jeżeli zaangażowanie Akcjonariusza Funkcjonalnego ustanie przed zwolnieniem wszystkich akcji, akcje niezwolnione podlegają zbyciu --- po cenie równej ich cenie emisyjnej --- na rzecz nabywcy wskazanego przez Radę Dyrektorów albo na rzecz Spółki, jeżeli nabycie akcji własnych jest dopuszczalne zgodnie z §~\ref{art:akcje} ust.~7. Akcje już zwolnione pozostają przy Akcjonariuszu Funkcjonalnym.
  \item Walne Zgromadzenie może, większością 75\% wszystkich głosów, zwolnić całość albo część akcji niezwolnionych z obowiązku zbycia, w szczególności w razie ustania zaangażowania wskutek śmierci, trwałej niezdolności do pracy, rozwiązania stosunku prawnego przez Spółkę bez istotnej przyczyny leżącej po stronie Akcjonariusza Funkcjonalnego albo odejścia uzgodnionego ze Spółką. Obowiązek zbycia nie wyłącza roszczeń Spółki o naprawienie szkody wyrządzonej naruszeniem obowiązków.
  \item Tryb wykonania obowiązku zbycia --- w tym sposób stwierdzenia ustania zaangażowania, obliczenie liczby akcji niezwolnionych, termin i sposób zapłaty, nieodwołalną ofertę zbycia oraz pełnomocnictwo do zawarcia umowy i wydania dyspozycji wpisu w rejestrze akcjonariuszy --- określa odrębna umowa wykonawcza, którą Akcjonariusz Funkcjonalny podpisuje najpóźniej w dniu objęcia akcji obowiązkiem zbycia. Pełnomocnictwo nie może być wykorzystane poza przypadkiem i procedurą określonymi w tej umowie.
  \item Obowiązek zbycia jest obowiązkiem związanym z akcją wobec Spółki i obciąża każdoczesnego posiadacza akcji nim objętych; podlega ujawnieniu w rejestrze akcjonariuszy zgodnie z §~\ref{art:akcje} ust.~5.
  \item W razie śmierci akcjonariusza jego spadkobiercy wstępują do Spółki. Akcje objęte obowiązkiem zbycia pozostają nim objęte; ust.~5 stosuje się odpowiednio.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Organy Spółki i Rada Dyrektorów}
\label{art:rada}
\begin{enumerate}
  \item Organami Spółki są Walne Zgromadzenie i Rada Dyrektorów. Spółka przyjmuje system monistyczny; nie powołuje się zarządu ani rady nadzorczej.
  \item Rada Dyrektorów prowadzi sprawy Spółki, reprezentuje Spółkę i sprawuje stały nadzór nad jej działalnością. Rada Dyrektorów składa się z 3 do 7 dyrektorów powoływanych i odwoływanych uchwałą Walnego Zgromadzenia podjętą bezwzględną większością głosów oddanych.
  \item Dyrektorów powołuje się na wspólną kadencję trwającą trzy lata obrotowe; dopuszczalne jest ponowne powołanie. Na podstawie art.~300$^{56}$ KSH mandat dyrektora wygasa z dniem powołania następcy albo z dniem podjęcia uchwały o niepowoływaniu następcy, nie później jednak niż z upływem 6 miesięcy od upływu kadencji. Mandat wygasa także wskutek śmierci, rezygnacji albo odwołania.
  \item Rada Dyrektorów wybiera ze swojego grona Przewodniczącego oraz wyznacza uchwałą co najmniej jednego dyrektora wykonawczego, któremu deleguje prowadzenie bieżącej działalności Spółki zgodnie z art.~300$^{76}$ KSH, zapewniając, aby co najmniej jeden dyrektor był dyrektorem niewykonawczym. Delegacja nie obejmuje spraw, o których mowa w art.~300$^{75}$ §~2--3 KSH ani w ust.~7, i może być w każdym czasie zmieniona albo cofnięta.
  \item Uchwały Rady Dyrektorów zapadają bezwzględną większością głosów przy udziale co najmniej połowy aktualnie powołanych dyrektorów. W razie równości głosów decyduje głos Przewodniczącego, z wyjątkiem spraw dotyczących jego interesu osobistego.
  \item Posiedzenia i głosowania Rady Dyrektorów mogą odbywać się na piśmie albo przy wykorzystaniu środków komunikacji elektronicznej, w tym poczty elektronicznej, jeżeli zapewniono identyfikację uczestników i utrwalenie treści głosów. Dyrektor, którego interes jest sprzeczny z interesem Spółki w danej sprawie, ujawnia to i nie uczestniczy w głosowaniu w tej sprawie; nie uwzględnia się go wówczas przy ustalaniu wymaganego udziału dyrektorów.
  \item Uchwały Rady Dyrektorów wymagają w szczególności: przyjęcie i zmiana rocznego budżetu oraz planu finansowego; zawarcie umowy albo dokonanie wydatku poza zatwierdzonym budżetem o wartości przekraczającej 250 000 zł; zaciągnięcie finansowania, udzielenie zabezpieczenia albo rozporządzenie aktywami o wartości przekraczającej 100 000 zł; zawarcie transakcji z akcjonariuszem, dyrektorem, osobą im bliską albo podmiotem powiązanym; utworzenie spółki zależnej albo rozpoczęcie działalności w nowej jurysdykcji; udzielenie licencji wyłącznej albo przeniesienie istotnej własności intelektualnej z zastrzeżeniem §~\ref{art:wz-zastrzezone}; przyjęcie polityk bezpieczeństwa informacji, kontroli eksportu i zgodności; wykonanie uchwał emisyjnych zgodnie z §~\ref{art:emisje} ust.~5; stwierdzenie wniesienia wkładów; wyrażenie albo odmowa zgody, o której mowa w §~\ref{art:zbywanie} ust.~1.
  \item Szczegółowy podział kompetencji, limity wydatków, zasady raportowania i politykę konfliktu interesów określa regulamin Rady Dyrektorów przyjęty jej uchwałą.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Reprezentacja Spółki}
\label{art:reprezentacja}
\begin{enumerate}
  \item Do składania oświadczeń w imieniu Spółki wymagane jest współdziałanie dwóch dyrektorów albo jednego dyrektora łącznie z prokurentem.
  \item Powołanie prokurenta wymaga zgody wszystkich dyrektorów; prokurę może odwołać każdy dyrektor zgodnie z art.~300$^{75}$ §~3 KSH.
  \item W umowie oraz w sporze między Spółką a dyrektorem Spółkę reprezentuje pełnomocnik powołany uchwałą akcjonariuszy albo --- w przypadku dyrektora wykonawczego --- dyrektor niewykonawczy działający na podstawie uchwały Rady Dyrektorów podjętej wyłącznie przez dyrektorów niewykonawczych, zgodnie z art.~300$^{79}$ §~3 KSH. Jeżeli wszystkie akcje przysługują jednemu akcjonariuszowi będącemu zarazem dyrektorem, stosuje się art.~300$^{79}$ §~4 KSH.
  \item Wewnętrzne limity kompetencji dyrektorów nie wpływają na skuteczność reprezentacji wobec osób trzecich.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Uchwały akcjonariuszy --- tryb, zdalne głosowanie i terminy}
\label{art:wz}
\begin{enumerate}
  \item Uchwały akcjonariuszy mogą być podejmowane na Walnym Zgromadzeniu albo poza Walnym Zgromadzeniem --- na piśmie albo przy wykorzystaniu środków komunikacji elektronicznej --- zgodnie z art.~300$^{80}$ §~1 KSH. Wybór trybu należy do zwołującego albo wnioskującego o podjęcie uchwały, z wyjątkiem uchwał, które z mocy prawa wymagają protokołu sporządzonego przez notariusza; te podejmuje się na Walnym Zgromadzeniu.
  \item \textbf{Komunikacja elektroniczna.} Zawiadomienia, projekty uchwał, karty do głosowania, oddane głosy i zawiadomienia o wyniku głosowania wysyła się pocztą elektroniczną na adres akcjonariusza wpisany do rejestru akcjonariuszy, a w braku takiego wpisu --- przesyłką poleconą na adres z rejestru. Zawiadomienie uważa się za dokonane z chwilą wysłania na adres wpisany do rejestru. Oświadczenia składane w wykonaniu niniejszego paragrafu wymagają formy dokumentowej; wystarcza wiadomość poczty elektronicznej wysłana z adresu wpisanego do rejestru akcjonariuszy.
  \item \textbf{Uchwały poza Walnym Zgromadzeniem.} Projekt uchwały wraz z uzasadnieniem i terminem oddania głosu rozsyła akcjonariuszom Rada Dyrektorów, Przewodniczący Rady Dyrektorów albo akcjonariusz reprezentujący co najmniej 5\% ogólnej liczby głosów. Termin oddania głosu nie może być krótszy niż 3 dni robocze od wysłania projektu i dłuższy niż 21 dni; za zgodą wszystkich akcjonariuszy termin może być krótszy. Uchwała zostaje podjęta, gdy w terminie oddano głosy „za” w liczbie wymaganej dla danej uchwały; niezwłocznie po upływie terminu albo po uzyskaniu wymaganej liczby głosów Przewodniczący Rady Dyrektorów stwierdza wynik głosowania i zawiadamia o nim akcjonariuszy pocztą elektroniczną. Głosu oddanego po terminie nie uwzględnia się. Do uchwał podejmowanych w tym trybie nie stosuje się wymogu tajnego głosowania (art.~300$^{80}$ §~3 KSH).
  \item \textbf{Walne Zgromadzenie.} Walne Zgromadzenia odbywają się w siedzibie Spółki albo w Warszawie. Zwołuje się je z zachowaniem terminów ustawowych, w sposób określony w ust.~2. Walne Zgromadzenie może podjąć uchwały bez formalnego zwołania, jeżeli reprezentowany jest cały kapitał akcyjny i nikt nie sprzeciwia się odbyciu Walnego Zgromadzenia ani wniesieniu poszczególnych spraw do porządku obrad.
  \item \textbf{Udział zdalny.} Dopuszcza się udział w Walnym Zgromadzeniu przy wykorzystaniu środków komunikacji elektronicznej zgodnie z art.~300$^{92}$ KSH, w szczególności przez dwustronną komunikację w czasie rzeczywistym i wykonywanie prawa głosu na odległość. Szczegółowe zasady udziału zdalnego, w tym sposób identyfikacji akcjonariusza, uzyskania dostępu do transmisji i oddania głosu, określa regulamin Walnego Zgromadzenia przyjęty uchwałą akcjonariuszy; Założyciele przyjmują pierwszy regulamin przy zawiązaniu Spółki. Regulamin nie może ustanawiać wymogów innych niż niezbędne do identyfikacji akcjonariuszy i zapewnienia bezpieczeństwa komunikacji elektronicznej.
  \item W zakresie dopuszczalnym przez prawo problemy techniczne leżące po stronie akcjonariusza nie wpływają na ważność Walnego Zgromadzenia ani podjętych uchwał.
  \item \textbf{Większości.} Każda akcja daje jeden głos. Uchwały zapadają bezwzględną większością głosów oddanych, chyba że ustawa albo niniejsza umowa wymaga większości surowszej. Większości określone w niniejszej umowie jako procent „wszystkich głosów w Spółce” oblicza się od pełnej liczby głosów istniejących w Spółce; jeżeli w danej sprawie akcjonariusz jest wyłączony od głosowania na podstawie bezwzględnie obowiązującego przepisu prawa, próg ten oblicza się od głosów nim nieobjętych. Niniejsza umowa nie wprowadza wyłączeń prawa głosu innych niż wynikające z ustawy.
  \item Dokumentację uchwał --- projekty, oddane głosy, zawiadomienia o wyniku i protokoły --- prowadzi i przechowuje Rada Dyrektorów w postaci elektronicznej, w sposób zapewniający jej integralność i dostępność dla każdego akcjonariusza.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Sprawy zastrzeżone dla Walnego Zgromadzenia}
\label{art:wz-zastrzezone}
\begin{enumerate}
  \item Uchwały w następujących sprawach wymagają większości 75\% wszystkich głosów w Spółce:
  \begin{enumerate}[label=\alph*)]
    \item zmiana niniejszej umowy, z zastrzeżeniem spraw wymagających większości surowszej na podstawie ustawy albo niniejszej umowy;
    \item emisja akcji, warrantów, opcji, instrumentów zamiennych albo innych praw do akcji, w tym każda emisja akcji serii~F i serii~P;
    \item nadanie albo cofnięcie statusu Założyciela Funkcjonalnego Serii~F;
    \item połączenie, podział, przekształcenie, rozwiązanie albo likwidacja Spółki;
    \item zbycie albo wydzierżawienie przedsiębiorstwa lub jego zorganizowanej części, ustanowienie na nich ograniczonego prawa rzeczowego oraz zbycie aktywów stanowiących ponad 25\% wartości aktywów Spółki;
    \item przeniesienie, zbycie albo udzielenie wyłącznej, nieodwołalnej lub bezterminowej licencji na kluczową własność intelektualną Spółki poza zwykłym tokiem działalności;
    \item nabycie akcji własnych, ich umorzenie albo wypłata z kapitału akcyjnego, jeżeli wymaga uchwały;
    \item zaciągnięcie zobowiązania, ustanowienie zabezpieczenia albo dokonanie transakcji poza zatwierdzonym budżetem o wartości przekraczającej 500 000 zł;
    \item zgoda na zbycie akcji w okresie, o którym mowa w §~\ref{art:zbywanie} ust.~4, oraz zwolnienie akcji z obowiązku zbycia na podstawie §~\ref{art:obowiazek-zbycia} ust.~5.
  \end{enumerate}
  \item Pozbawienie prawa poboru wymaga większości co najmniej czterech piątych głosów (§~\ref{art:emisje} ust.~6).
  \item Pozostałe uchwały akcjonariuszy, w tym powołanie i odwołanie dyrektorów, zatwierdzenie sprawozdania finansowego, udzielenie absolutorium, przeznaczenie zysku i ustalenie wynagrodzenia dyrektorów, zapadają bezwzględną większością głosów oddanych.
  \item Uchwały Walnego Zgromadzenia w sprawach zastrzeżonych ustawą do kompetencji Rady Dyrektorów, w szczególności w sprawach z art.~300$^{75}$ §~2 KSH, stanowią zgodę na dokonanie czynności i nie zastępują uchwały Rady Dyrektorów. Na podstawie art.~300$^{81}$ pkt~3 KSH nabycie i zbycie nieruchomości, użytkowania wieczystego albo udziału w nieruchomości o wartości nieprzekraczającej progu z ust.~1 lit.~h nie wymaga uchwały Walnego Zgromadzenia.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Własność intelektualna, poufność i bezpieczeństwo}
\label{art:ip}
\begin{enumerate}
  \item Prawa własności intelektualnej powstałe w związku z działalnością Spółki, w tym autorskie prawa majątkowe do oprogramowania i dokumentacji, prawa do wynalazków, wzorów i baz danych, przysługują Spółce. Każdy Założyciel, dyrektor, pracownik i współpracownik zawiera ze Spółką umowę zapewniającą nabycie przez Spółkę tych praw, w tym praw zależnych i prawa do zezwalania na wykonywanie praw zależnych, oraz przeniesienie nośników i dokumentacji.
  \item Założyciel, który przed zawiązaniem Spółki wytworzył albo nabył technologię wykorzystywaną przez Spółkę, a nieobjętą wkładem, udziela Spółce nieodpłatnej, niewyłącznej, nieograniczonej terytorialnie licencji na potrzeby jej działalności i ujawnia Radzie Dyrektorów prawa osób trzecich do tej technologii.
  \item Akcjonariusze i dyrektorzy zachowują w poufności informacje Spółki o wartości gospodarczej, w tym kod, modele, dane, dokumentację techniczną, informacje handlowe i warunki finansowania, oraz przestrzegają polityk bezpieczeństwa informacji przyjętych przez Radę Dyrektorów. Obowiązek ten trwa także po ustaniu zaangażowania albo zbyciu akcji, przez okres i w zakresie dopuszczalnym przez prawo.
  \item Spółka, jej akcjonariusze i dyrektorzy przestrzegają przepisów o sankcjach, środkach ograniczających i kontroli obrotu towarami, technologiami i usługami o znaczeniu strategicznym. Udostępnienie technologii objętej kontrolą eksportu, w tym akcjonariuszowi i dyrektorowi, następuje wyłącznie zgodnie z tymi przepisami i polityką kontroli eksportu przyjętą przez Radę Dyrektorów.
  \item Akcjonariusz i dyrektor nie prowadzi działalności konkurencyjnej wobec Spółki ani nie uczestniczy w podmiocie konkurencyjnym bez zgody Rady Dyrektorów wyrażonej uchwałą podjętą bez udziału osoby zainteresowanej; obowiązek ten nie dotyczy inwestycji portfelowych nieprzekraczających 5\% udziału i niewiążących się z dostępem do informacji poufnych albo z funkcją w organach.
  \item Naruszenie obowiązków, o których mowa w niniejszym paragrafie, zobowiązuje do niezwłocznego zaniechania naruszenia, usunięcia jego skutków i naprawienia szkody; nie wyłącza to innych środków przewidzianych prawem albo odrębnymi umowami.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Zysk i wypłaty}
\label{art:zysk}
\begin{enumerate}
  \item O przeznaczeniu zysku decyduje Walne Zgromadzenie, z uwzględnieniem ograniczeń ustawowych, potrzeb finansowania rozwoju, zobowiązań grantowych i bezpieczeństwa płynności. W pierwszych 36 miesiącach od wpisu Spółki do rejestru rekomendowanym sposobem wykorzystania zysku jest jego reinwestowanie.
  \item Wypłata dywidendy, zaliczki na dywidendę, wypłata z kapitału akcyjnego albo inne świadczenie na rzecz akcjonariuszy może nastąpić wyłącznie po przeprowadzeniu wymaganych prawem testów i procedur, w szczególności testu wypłacalności z art.~300$^{15}$ KSH, oraz zgodnie z warunkami umów finansowania.
  \item Spółka tworzy wymagane prawem odpisy na kapitał akcyjny i może tworzyć kapitały rezerwowe, w tym kapitał rezerwowy na nabycie akcji własnych.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Rozwiązanie i likwidacja Spółki}
\label{art:likwidacja}
\begin{enumerate}
  \item Rozwiązanie Spółki następuje w przypadkach przewidzianych prawem albo na podstawie uchwały Walnego Zgromadzenia podjętej większością 75\% wszystkich głosów.
  \item Likwidatorami są dyrektorzy, chyba że Walne Zgromadzenie postanowi inaczej.
  \item Przed wydaniem albo zniszczeniem nośników, prototypów i dokumentacji likwidatorzy dokonują klasyfikacji bezpieczeństwa i eksportowej oraz zapewniają zgodne z prawem zabezpieczenie własności intelektualnej Spółki.
  \item Majątek pozostały po zaspokojeniu albo zabezpieczeniu wierzycieli dzieli się między akcjonariuszy proporcjonalnie do liczby akcji.
\end{enumerate}
\end{ContractClause}

\begin{ContractClause}{Postanowienia końcowe}
\label{art:final}
\begin{enumerate}
  \item Zmiana niniejszej umowy wymaga formy przewidzianej prawem i większości określonej w §~\ref{art:wz-zastrzezone} ust.~1 lit.~a. Zmiana umowy zwiększająca obowiązki związane z akcją wobec akcjonariusza, w szczególności zaostrzająca zasady z §~\ref{art:zbywanie} ust.~4 albo §~\ref{art:obowiazek-zbycia}, wymaga zgody każdego akcjonariusza, którego dotyczy.
  \item Jeżeli postanowienie umowy jest nieważne albo niewykonalne, pozostałe postanowienia zachowują moc; strony zastąpią wadliwe postanowienie rozwiązaniem prawnie dopuszczalnym, możliwie najbliższym jego celowi gospodarczemu.
  \item Odesłania do przepisów prawa rozumie się jako odesłania do przepisów w brzmieniu obowiązującym w dniu dokonania danej czynności, a w razie ich uchylenia albo zmiany numeracji --- do przepisów, które je zastąpiły. Rada Dyrektorów przedstawia Walnemu Zgromadzeniu projekt dostosowania umowy w terminie 6 miesięcy od dnia wejścia w życie zmiany przepisów wymagającej zmiany umowy.
  \item Spory związane z niniejszą umową strony w pierwszej kolejności poddają negocjacjom i mediacji; w razie braku ugody właściwy jest sąd powszechny właściwy dla siedziby Spółki. Obowiązek ten nie ogranicza prawa do żądania zabezpieczenia ani do dokonania czynności niecierpiącej zwłoki.
  \item Załącznik nr~1 stanowi integralną część niniejszej umowy. Zmiana danych osobowych lub adresowych Założyciela, zbycie akcji oraz stwierdzenie wniesienia wkładu nie wymagają zmiany Załącznika nr~1 i są ujawniane w rejestrze akcjonariuszy albo uchwałą Rady Dyrektorów.
  \item Pierwszych dyrektorów, podmiot prowadzący rejestr akcjonariuszy oraz pierwszy regulamin Walnego Zgromadzenia wskazują Założyciele w uchwałach objętych aktem zawiązania Spółki. Do czasu wpisu Spółki do rejestru przedsiębiorców Spółkę w organizacji reprezentuje Rada Dyrektorów zgodnie z §~\ref{art:reprezentacja}, a zaciąganie zobowiązań innych niż niezbędne do zawiązania i rejestracji Spółki wymaga zgody wszystkich Założycieli.
\end{enumerate}
\end{ContractClause}
